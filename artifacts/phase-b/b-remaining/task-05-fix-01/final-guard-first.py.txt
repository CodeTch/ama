from pathlib import Path
import json,subprocess,datetime,hashlib,socket,os
r=Path('/home/tanren/ama/.worktrees/phase-b-completion')
w=r/'.superpowers/sdd/2026-10-23-ama-phase-b-remaining';e=r/'artifacts/phase-b/b-remaining/task-05-fix-01'
stamp=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()
proof=json.loads((w/'resume-20261024-root-selected-resources.json').read_text())
fields={'id':'.Id','name':'.Name','status':'.State.Status','running':'.State.Running','oom':'.State.OOMKilled','image':'.Config.Image','memory':'.HostConfig.Memory','swap':'.HostConfig.MemorySwap','nano_cpus':'.HostConfig.NanoCpus','pids':'.HostConfig.PidsLimit','ports':'.HostConfig.PortBindings','mounts':'.Mounts'}
template='{'+','.join('"'+key+'":{{json '+field+'}}' for key,field in fields.items())+'}'
guard=[]
for original in proof['containers']:
 cid=original['requested_id'];label='ama.remaining.owner' if cid.startswith('eefdd') else 'ama.owner'
 selected_template=template[:-1]+',"owner":{{json (index .Config.Labels "'+label+'")}}}'
 cmd=['docker','inspect','--format',selected_template,cid]
 result=subprocess.run(cmd,capture_output=True,text=True,check=True);actual=json.loads(result.stdout)
 expected={k:original['selected'][k] for k in fields};expected['owner']=original['selected']['labels'][label]
 changes={k:{'expected':expected[k],'actual':actual[k]} for k in expected if actual[k]!=expected[k]}
 guard.append({'exactCID':cid,'command':cmd,'selected':actual,'changes_from_resume':changes})
 assert not changes, 'selected resource identity changed'
(e/'final-selected-resources.json').write_text(json.dumps({'utc':stamp(),'method':'explicit selected --format, no Config.Env/full inspect','containers':guard},indent=2)+'\n')

owned={}
for p in e.glob('case-*-native-started.json'):
 data=json.loads(p.read_text());owned[data['pid']]={'kind':'native','configuration':data['command'][-1],'start':data.get('startInstant')}
 owned[data['parentPid']]={'kind':'test-jvm','tree':str(r)}
processes=[]
for pid,expected in sorted(owned.items()):
 path=Path('/proc')/str(pid)
 if not path.exists():processes.append({'pid':pid,'kind':expected['kind'],'present':False});continue
 command=(path/'cmdline').read_bytes().replace(b'\0',b' ').decode(errors='replace')
 same=expected.get('configuration','__NO_CONFIGURATION__') in command if expected['kind']=='native' else str(r) in os.readlink(path/'cwd')
 processes.append({'pid':pid,'kind':expected['kind'],'present':True,'matches_owned_command_or_cwd':same})
 assert not same, 'exact owned JVM remains active'
(e/'final-known-processes.json').write_text(json.dumps({'utc':stamp(),'method':'only exact recorded native/test PIDs; no global process enumeration','processes':processes},indent=2)+'\n')

source=e/'final-local-lifetime-custody/source.json';start=source.stat().st_mtime
end=datetime.datetime.fromisoformat(json.loads((e/'final-local-lifetime-custody/command.json').read_text())['utc']).timestamp()
orders=[]
for case in e.glob('pipeline-*'):
 resource=case/'resource.json'
 if not resource.exists() or not start<=resource.stat().st_mtime<=end:continue
 data=json.loads(resource.read_text());port=data['nativePort']
 paths=sorted(case.glob('maintenance-*.json'))+[case/'loop-close.json']
 assert all(p.exists() for p in paths)
 pre=next(case.glob('maintenance-prelinked-*.json'));after=next(case.glob('maintenance-after-*.json'));closed=case/'loop-close.json'
 assert pre.stat().st_mtime<=after.stat().st_mtime<=closed.stat().st_mtime
 sock=socket.socket();sock.settimeout(.2)
 try:code=sock.connect_ex(('127.0.0.1',port))
 finally:sock.close()
 assert code!=0,'owned final route still accepts TCP'
 orders.append({'case':case.name,'nativePid':data['nativePid'],'port':port,'observedPortClosedUtc':stamp(),'connectExit':code,'file_times':[{'path':str(p.relative_to(r)),'mtimeUtc':datetime.datetime.fromtimestamp(p.stat().st_mtime,datetime.timezone.utc).isoformat()} for p in [pre,after,closed]],'source_order':'loop.close join -> preparation exit/active=0 -> guarded maintenance -> loop-close receipt -> finally server.stop -> @AfterEach native receipt','qualification':'file times plus executed source order; port closure observed now, no invented historical server.stop timestamp'})
assert len(orders)==3,len(orders)
(e/'final-local-custody-order.json').write_text(json.dumps({'utc':stamp(),'sourceSha256':hashlib.sha256((r/'backend/integration-tests/src/test/java/ama/storage/Task05FixOpenIT.java').read_bytes()).hexdigest(),'cases':orders},indent=2)+'\n')
print('selected resources unchanged; exact known JVMs gone; final LOCAL3 custody source/file times and closed ports verified')
