import os, pathlib, subprocess, json, shlex, hashlib, sys
root=pathlib.Path(__file__).resolve().parents[2]
e=root/'artifacts/phase-b/b-remaining/task-01'
env=os.environ.copy()
for line in (root/'.tools/phase-b-remaining/env.sh').read_text().splitlines():
    key,value=line.removeprefix('export ').split('=',1); env[key]=shlex.split(value)[0]
jdk='/home/tanren/ama/.worktrees/phase-a/.tools/zulu21.52.203-ca-jdk21.0.12.1-linux_x64'
env['JAVA_HOME']=jdk; env['PATH']=jdk+'/bin:'+env['PATH']
name=sys.argv[1]; args=sys.argv[2:]
files=[p for folder in ['backend/core/src','backend/adapters/src','backend/worker/src','backend/integration-tests/src'] for p in (root/folder).rglob('*') if p.is_file()]
(e/(name+'.sources.json')).write_text(json.dumps({str(p.relative_to(root)):hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(files)},indent=2)+'\n')
r=subprocess.run(args,cwd=root,env=env,capture_output=True,text=True)
log=r.stdout+r.stderr
for k in ['AMA_TEST_DATABASE_PASSWORD']: log=log.replace(env[k],'[REDACTED]')
(e/(name+'.log')).write_text(log)
(e/(name+'.command.json')).write_text(json.dumps({'argv':args,'cwd':str(root),'exit_code':r.returncode,'JAVA_HOME':jdk,'private_env_path':'.tools/phase-b-remaining/env.sh'},indent=2)+'\n')
print(log[-12000:]);sys.exit(r.returncode)
